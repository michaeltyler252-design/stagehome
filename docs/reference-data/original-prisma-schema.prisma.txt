// ============================================================================
// Student Housing Marketplace — Canonical Nationwide Data Model
// Phase 1, Milestone 1: Project Foundation
//
// Scope note (read before extending):
// This schema implements every table group listed in the master spec Part F
// ("Geography and education", "Identity and organisations", "Property
// inventory", "Commercial terms", "Booking and payments", "Agreements",
// "Engagement and operations"). Tables are NATIONWIDE and reusable — no
// county-specific tables exist, per Part F instruction.
//
// Data-quality fields (source_status, verification_status, source_file,
// source_record_reference, source_url, verified_at, verified_by,
// confidence_level, conflict_status, publication_status, notes) are applied
// in full to every entity that can be populated from SOURCE_SUPPLIED_UNVERIFIED
// external data (geography, education, and property-inventory groups).
// Operational/transactional tables (bookings, payments, agreements, etc.)
// carry status + audit timestamps but not the external-source fields, since
// they are only ever created internally by the platform, never imported.
//
// Field-level depth for booking, payment, agreement, and engagement tables
// is intentionally minimal in Milestone 1. Full column-level design for
// those tables is Milestone 2 ("Database and seed data") scope, once Nairobi
// verification and seed strategy are approved.
// ============================================================================

generator client {
  provider        = "prisma-client-js"
  previewFeatures = ["postgresqlExtensions", "multiSchema"]
}

datasource db {
  provider   = "postgresql"
  url        = env("DATABASE_URL")
  extensions = [postgis, pg_trgm]
  schemas    = ["public", "staging"]
}

// ============================================================================
// STAGING SCHEMA — raw source-data import
// See docs/data-quality/source-import-staging-design.md
//
// Nothing in this group is ever read by apps/web or any public API route.
// Only internal ETL jobs (packages/database/src/staging/*) and the future
// administrator "data-quality queue" (Part K) read from it. Promotion to the
// canonical `public` schema tables above is a separate, explicit, logged step
// — never automatic.
// ============================================================================

model RawImportBatch {
  id           String   @id @default(cuid())
  batchKey     String   @unique // e.g. "nairobi-phase1-milestone2-audit"
  checksum     String   // sha256 of the manifest + all referenced raw text files
  county       String?
  importedBy   String?
  createdAt    DateTime @default(now())

  propertyRecords   RawPropertyRecord[]
  universityRecords RawUniversityRecord[]
  fieldIssues       RawFieldIssue[]

  @@schema("staging")
  @@map("raw_import_batches")
}

model RawPropertyRecord {
  id                    String          @id @default(cuid())
  batchId               String
  batch                 RawImportBatch  @relation(fields: [batchId], references: [id])
  propertyName           String
  universityName          String? // as supplied, not yet resolved to a canonical University row
  sourceFile             String?
  rawText                 String          // verbatim text preserved from the source document
  conflictStatus          String          @default("NONE") // NONE | FLAGGED | UNDER_REVIEW | RESOLVED
  promotedPropertyId      String? // set once/if promoted to public.properties
  createdAt               DateTime        @default(now())

  @@index([propertyName])
  @@schema("staging")
  @@map("raw_property_records")
}

model RawUniversityRecord {
  id                  String         @id @default(cuid())
  batchId             String
  batch               RawImportBatch @relation(fields: [batchId], references: [id])
  universityName      String
  campusName          String?
  sourceFile          String?
  rawExcerpt          String?
  promotedUniversityId String? // set once/if promoted to public.universities
  createdAt           DateTime       @default(now())

  @@index([universityName])
  @@schema("staging")
  @@map("raw_university_records")
}

model RawFieldIssue {
  id           String         @id @default(cuid())
  batchId      String
  batch        RawImportBatch @relation(fields: [batchId], references: [id])
  recordType   String // property | university
  recordRef    String // propertyName or universityName as supplied
  issue        String
  createdAt    DateTime       @default(now())

  @@schema("staging")
  @@map("raw_field_issues")
}
